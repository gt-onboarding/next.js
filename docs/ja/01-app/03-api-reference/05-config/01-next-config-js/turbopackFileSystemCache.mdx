## 使用方法

Turbopack FileSystem Cache によって、`next dev` や `next build` コマンド間で Turbopack の処理量を削減できます。有効化すると、Turbopack はビルド間で `.next` フォルダにデータを保存および復元し、これにより以降のビルドや開発セッションを大幅に高速化できます。

> **補足:** FileSystem Cache 機能は開発用途では安定版とみなされますが、本番ビルドでは実験的な機能とみなされています。

```ts filename="next.config.ts" switcher
import type { NextConfig } from 'next'

const nextConfig: NextConfig = {
  experimental: {
    // `next dev` のファイルシステムキャッシュを有効にする
    turbopackFileSystemCacheForDev: true,
    // `next build` のファイルシステムキャッシュを有効にする
    turbopackFileSystemCacheForBuild: true,
  },
}

export default nextConfig
```

```js filename="next.config.js" switcher
/** @type {import('next').NextConfig} */
const nextConfig = {
  experimental: {
    // `next dev` のファイルシステムキャッシュを有効にする
    turbopackFileSystemCacheForDev: true,
    // `next build` のファイルシステムキャッシュを有効にする
    turbopackFileSystemCacheForBuild: true,
  },
}

module.exports = nextConfig
```
